4.Remove Empty Collections

Indentiy empty collections

db.getCollectionNames().forEach(function (collection) {
    if (db[collection].countDocuments() === 0) {
        print(collection);
    }
});
Remove empty collections

db.getCollectionNames().forEach(function (collection) {
    if (db[collection].countDocuments() === 0) {
        db[collection].drop();
    }
});
